package com.example.game3315298.game

import androidx.compose.foundation.Canvas
import androidx.compose.foundation.gestures.detectDragGestures
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.Row
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Button
import androidx.compose.material3.Text
import androidx.compose.runtime.*

import androidx.compose.ui.Modifier
import androidx.compose.ui.geometry.Offset
import androidx.compose.ui.geometry.Size
import androidx.compose.ui.graphics.Brush
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.Path
import androidx.compose.ui.input.pointer.pointerInput
import androidx.compose.ui.layout.onSizeChanged
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import kotlinx.coroutines.delay

@Composable
fun GameScreen() {
    var gameStarted by remember { mutableStateOf(false) }

    Box(modifier = Modifier.fillMaxSize()) {
        var playerX by remember { mutableFloatStateOf(0f) }
        var playerY by remember { mutableFloatStateOf(0f) }
        var playerInitialized by remember { mutableStateOf(false) }
        var screenWidth by remember { mutableFloatStateOf(0f) }

        // Estados con los agregados del tercer código (formación ampliada y movimiento)
        var enemies by remember { mutableStateOf(createEnemyFormation()) }
        var enemyDirection by remember { mutableStateOf(1f) }
        var score by remember { mutableIntStateOf(0) }
        var lives by remember { mutableIntStateOf(3) }
        var bullets by remember { mutableStateOf(listOf<Bullet>()) }

        // Bucle para el desplazamiento y limpieza de balas (del tercer código)
        LaunchedEffect(Unit) {
            while (true) {
                delay(16L)
                bullets = bullets.map { bullet ->
                    bullet.copy(y = bullet.y - 8f)
                }.filter { bullet -> bullet.y > 0f }
            }
        }

        // Bucle para el movimiento autónomo de los enemigos con rebote en bordes (del tercer código)
        LaunchedEffect(Unit) {
            while (true) {
                delay(30L)
                val enemyWidth = 60f
                val margin = enemyWidth / 2
                val speed = 3f
                val leftEdge = enemies.minOfOrNull { it.x - margin } ?: 0f
                val rightEdge = enemies.maxOfOrNull { it.x + margin } ?: 0f

                if ((enemyDirection > 0f && rightEdge + speed >= screenWidth) ||
                    (enemyDirection < 0f && leftEdge - speed <= 0f)) {
                    enemyDirection *= -1f
                }

                enemies = enemies.map { enemy ->
                    enemy.copy(x = enemy.x + (speed * enemyDirection))
                }
            }
        }

        Canvas(
            modifier = Modifier
                .fillMaxSize()
                .onSizeChanged { size ->
                    screenWidth = size.width.toFloat()
                    if (!playerInitialized) {
                        playerX = size.width / 2f
                        playerY = size.height - 220f
                        playerInitialized = true
                    }
                }
                .pointerInput(Unit) {
                    detectDragGestures(
                        onDrag = { change, dragAmount ->
                            change.consume()
                            val shipWidth = 80f
                            playerX = (playerX + dragAmount.x)
                                .coerceIn(shipWidth / 2, screenWidth - shipWidth / 2)
                        }
                    )
                }
        ) {
            drawRect(color = Color.Black)

            // Se mantiene la nave original (Triángulo clásico)
            val shipWidth = 80f
            val shipHeight = 80f
            val left = playerX - shipWidth / 2
            val top = playerY - shipHeight / 2
            val right = playerX + shipWidth / 2
            val bottom = playerY + shipHeight / 2
            val shipPath = Path().apply {
                moveTo(playerX, top)
                lineTo(left, bottom)
                lineTo(playerX, bottom - 15f)
                lineTo(right, bottom)
                close()
            }
            drawPath(shipPath, color = Color.White)

            // Dibujar Enemigos con el efecto de degradado (del tercer código)
            enemies.forEach { enemy ->
                val enemyWidth = 60f
                val enemyHeight = 40f
                drawRect(
                    brush = Brush.linearGradient(
                        colors = listOf(Color(0xFFC0C0C0), Color(0xFFFF0000), Color(0xFF052AF3)),
                        start = Offset(enemy.x - enemyWidth / 2, enemy.y - enemyHeight / 2),
                        end = Offset(enemy.x + enemyWidth / 2, enemy.y + enemyHeight / 2)
                    ),
                    topLeft = Offset(
                        enemy.x - enemyWidth / 2f,
                        enemy.y - enemyHeight / 2f
                    ),
                    size = Size(enemyWidth, enemyHeight)
                )
            }

            // Dibujar Balas desplazándose
            bullets.forEach { bullet ->
                drawRect(
                    color = Color.Yellow,
                    topLeft = Offset(bullet.x - 4f, bullet.y - 15f),
                    size = Size(8f, 20f)
                )
            }
        }

        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(top = 100.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceEvenly
            ) {
                Text(
                    text = "SCORE: $score",
                    color = Color.White,
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Bold
                )

                Text(
                    text = "LIVES " + "❤️".repeat(lives.coerceAtLeast(0)),
                    color = Color.White,
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Bold
                )
            }

            Text(
                text = "SPACE SHOOTER",
                color = Color.White,
                fontSize = 28.sp,
                fontWeight = FontWeight.Bold,
                modifier = Modifier.padding(vertical = 8.dp)
            )

            Button(onClick = {
                gameStarted = true
                score += 100
            }) {
                Text(text = "START")
            }

            if (gameStarted) {
                Text(text = "GAME STARTED", color = Color.White)
            }

            Button(onClick = {
                bullets = bullets + Bullet(x = playerX, y = playerY - 40f)
            }) {
                Text(text = "FIRE")
            }
        }
    }
}

// Formación ampliada de 6 enemigos (del tercer código)
fun createEnemyFormation(): List<Enemy> {
    return listOf(
        Enemy(150f, 150f),
        Enemy(300f, 150f),
        Enemy(450f, 150f),
        Enemy(600f, 150f),

    )
}
